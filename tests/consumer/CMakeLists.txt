cmake_minimum_required(VERSION 4.3.3)
if(CMAKE_VERSION VERSION_LESS 4.4)
  set(CMAKE_EXPERIMENTAL_CXX_IMPORT_STD "451f2fe2-a8a2-47c3-bc32-94786d8fc91b")
else()
  set(CMAKE_EXPERIMENTAL_CXX_IMPORT_STD "f35a9ac6-8463-4d38-8eec-5d6008153e7d")
endif()
set(CMAKE_CXX_STANDARD 26)
set(CMAKE_CXX_MODULE_STD ON)
project(serial_xml_consumer LANGUAGES CXX)

if(SERIAL_XML_SOURCE_DIR)
  include(FetchContent)
  FetchContent_Declare(serial_xml SOURCE_DIR "${SERIAL_XML_SOURCE_DIR}")
  FetchContent_MakeAvailable(serial_xml)
  if(BUILD_EXAMPLES OR BUILD_BENCHMARKS)
    message(FATAL_ERROR "Embedded builds must default to library-only")
  endif()
else()
  find_package(SerialXML CONFIG REQUIRED)
endif()

add_executable(consumer main.cpp)
target_link_libraries(consumer PRIVATE serial_xml::serial_xml)

foreach(case RANGE 1 6)
  add_executable(invalid_name_${case} EXCLUDE_FROM_ALL invalid_name.cpp)
  target_compile_definitions(invalid_name_${case}
                             PRIVATE INVALID_NAME_CASE=${case})
  target_link_libraries(invalid_name_${case} PRIVATE serial_xml::serial_xml)
endforeach()
